      ```javascript
// GitHub webhook handler
const handlePush = async (payload) => {
  if (payload.ref === 'refs/heads/main') {
    await updateDocumentation({
      repo: payload.repository.name,
      files: payload.commits[0].modified
    });
  }
};
```

<CodeGroup tabs="GitHub Actions,Jenkins">
  ```yaml
# .github/workflows/docs.yml
name: Update Documentation
on:
  push:
    branches: [ main ]
    paths: [ 'docs/**' ]

jobs:
  update-docs:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v3
      - name: Sync to Ismet Docs
        run: |
          curl -X POST https://api.ismetdocs.com/v1/sync \
            -H "Authorization: Bearer ${{ secrets.DOCS_API_KEY }}" \
            -H "Content-Type: application/json" \
            -d '{"repo": "my-project", "branch": "main"}'
```
  ```groovy
// Jenkins pipeline
pipeline {
    agent any
    stages {
        stage('Update Docs') {
            steps {
                sh '''
                    curl -X POST https://api.ismetdocs.com/v1/sync \
                      -H "Authorization: Bearer $DOCS_API_KEY" \
                      -H "Content-Type: application/json" \
                      -d '{"repo": "my-project", "branch": "main"}'
                '''
            }
        }
    }
}
```
</CodeGroup>
